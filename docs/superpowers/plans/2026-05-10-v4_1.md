```python
# src/cost.py
"""Cost accumulator — token-usage to USD with a documented price table.

Pure functions. No global state. Reads data/model_prices.json once at module
load. Unknown models return 0.0 and log a warning rather than crash —
honest "unknown cost" beats a fake number.
"""

from __future__ import annotations

import json
import logging
from pathlib import Path
from typing import Any

logger = logging.getLogger(__name__)

_PRICES_PATH = Path(__file__).parent.parent / "data" / "model_prices.json"


def _load_prices() -> dict[str, dict[str, float]]:
    if not _PRICES_PATH.exists():
        logger.warning("model_prices.json missing — all costs will return 0.0")
        return {}
    raw = json.loads(_PRICES_PATH.read_text(encoding="utf-8"))
    return {k: v for k, v in raw.items() if not k.startswith("_")}


_PRICES = _load_prices()


def usage_to_usd(usage: dict[str, int] | Any, model_id: str) -> float:
    """Convert {'prompt_tokens': N, 'completion_tokens': M} to USD for `model_id`.

    Returns 0.0 for unknown models with a logged warning. Never raises —
    cost tracking is observability, not a production gate.
    """
    if model_id not in _PRICES:
        logger.warning("Unknown model %s — cost defaulting to 0.0", model_id)
        return 0.0
    prices = _PRICES[model_id]
    prompt = (usage.get("prompt_tokens") if isinstance(usage, dict) else getattr(usage, "prompt_tokens", 0)) or 0
    completion = (usage.get("completion_tokens") if isinstance(usage, dict) else getattr(usage, "completion_tokens", 0)) or 0
    cost = (prompt * prices["input_per_mtok"] + completion * prices["output_per_mtok"]) / 1_000_000.0
    return float(cost)


def accumulate_cost(
    prior_total: float,
    prior_breakdown: dict[str, float],
    agent: str,
    usage: dict[str, int] | Any,
    model_id: str,
) -> dict[str, Any]:
    """Build a state-update dict adding this call's cost to the running total.

    Caller merges the returned dict into AgentState. `prior_breakdown` is
    copied (not mutated) so audit-log invariants hold.
    """
    cost = usage_to_usd(usage, model_id)
    new_breakdown = dict(prior_breakdown)
    new_breakdown[agent] = new_breakdown.get(agent, 0.0) + cost
    return {
        "total_cost_usd": prior_total + cost,
        "cost_breakdown": new_breakdown,
    }


__all__ = ["accumulate_cost", "usage_to_usd"]
```

```python
# eval/run_experiments.py — additions

from types import SimpleNamespace
from eval.multiagent_evaluators import (
    tool_selection_precision,
    critic_disagreement_with_drafter,
    critic_alignment_with_humans,
    loop_iteration_count,
    agent_cost_breakdown,
)

# Inside the per-ticket loop, after final state captured:
def _ma_evaluators_for(ticket: dict, state: dict) -> dict[str, float]:
    """Run all 5 multi-agent evaluators against this ticket's final state."""
    run_obj = SimpleNamespace(outputs=state)
    example_obj = SimpleNamespace(outputs={"expected_intent": ticket.get("expected_intent", "")})
    out = {}
    for fn in (tool_selection_precision, critic_disagreement_with_drafter,
               critic_alignment_with_humans, loop_iteration_count, agent_cost_breakdown):
        result = fn(run_obj, example_obj)
        out[result["key"]] = result["score"]
    return out

# After all tickets done — aggregate:
def _aggregate_ma(per_ticket_ma: list[dict]) -> dict[str, float]:
    if not per_ticket_ma:
        return {}
    keys = per_ticket_ma[0].keys()
    return {f"{k}_avg": sum(p[k] for p in per_ticket_ma) / len(per_ticket_ma) for k in keys}
```

Wire `_ma_evaluators_for(ticket, final_state)` into the per-ticket loop, store results in a list, then call `_aggregate_ma(...)` at the end and merge into the top-level `results` dict before writing JSON.
